## Duplicate-ID check

Command:
```bash
python - <<'PY'
from pathlib import Path
import re
from collections import Counter
import yaml
ids=[]
for path in Path('/mnt/apps/ForLLm').rglob('*.md'):
    if any(part.startswith('.') for part in path.parts):
        continue
    if path.name in {'README.md','schema.md','CONTRIBUTING.md','RELEASE_CHECKLIST.md','CHANGELOG.md','LICENSE','LLM_CODEBASE_KNOWLEDGE_BASE.md'}:
        continue
    if any(p.name in {'templates','.github','docs','architecture','scripts','benchmark_prompts','prompts','build','__pycache__','node_modules','references','llm_kb'} for p in path.parents):
        continue
    m=re.match(r'^---\n(.*?)\n---', path.read_text(), re.S)
    if m:
        meta=yaml.safe_load(m.group(1)) or {}
        if 'id' in meta:
            ids.append((meta['id'], str(path.relative_to('/mnt/apps/ForLLm'))))
counts=Counter(i for i,_ in ids)
dups=[(i,p) for i,p in ids if counts[i]>1]
for i,p in dups:
    print(f'{i}\t{p}')
print(f'duplicate_ids={len(dups)}')
PY
```

Result:
```text
duplicate_ids=0
```
